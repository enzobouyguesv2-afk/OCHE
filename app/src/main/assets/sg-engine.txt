    }
    choicesFor(sc) {
      return (sc.choices || []).filter(c => !(c.once && this.st.used[this.st.scene + ":" + c.id]) && this.cond(c.req) && (!c.cost || Object.keys(c.cost).every(k => k === "menace" || this.st.res[k] + c.cost[k] >= 0)));
    }
    lockedChoices(sc) {
      return (sc.choices || []).filter(c => !(c.once && this.st.used[this.st.scene + ":" + c.id]) && c.showLocked && !this.choicesFor(sc).includes(c));
    }
    afterNarr(now) {
      const S = this.st, sc = this.sc();
      if (sc.puzzle) { S.step = { kind: "puzzle", attempts: Math.max(1, (sc.puzzle.attempts || 3) + (this.diff().att - 3)), max: 0, tries: [], until: now + this.dur().puzzle, len: this.dur().puzzle, minAt: 0 }; S.step.max = S.step.attempts; return; }
      const ch = this.choicesFor(sc);
      if (ch.length) { S.step = { kind: "vote", choices: ch.map(c => c.id), votes: {}, ordre: null, until: now + this.dur().vote, len: this.dur().vote, minAt: now + 6000 }; return; }
      if (sc.auto) return this.showOutcome(this.pick(sc.auto), { now, label: null });
      this.goNext(sc.next || "END", now);
    }
    chance(c) {
      if (!c.check) return null;
      const k = c.check; let p = k.chance + this.diff().chk;
      if (k.item && this.st.items[k.item]) p += k.itemBonus || 0.2;
      if (k.flag && this.st.flags[k.flag]) p += k.flagBonus || 0.2;
      return p;
    }
    vote(pid, choice, now) {
      const S = this.st, s = S.step;
      if (!s || s.kind !== "vote") return { ok: false, reason: "ferme" };
      if (!s.choices.includes(choice)) return { ok: false, reason: "option" };
      s.votes[pid] = choice; this.bump(); this.checkAll(now); return { ok: true };
    }
    closeVote(now) {
      const S = this.st, s = S.step, sc = this.sc(), tally = {};
      s.choices.forEach(c => tally[c] = 0);
      Object.values(s.votes).forEach(c => { if (tally[c] != null) tally[c]++; });
      let winner = null, how = "majorite";
      const chefVote = s.ordre && s.votes[s.ordre];
      if (chefVote) { winner = chefVote; how = "ordre"; }
      else {
        const top = Math.max.apply(null, Object.values(tally)), tied = s.choices.filter(c => tally[c] === top);
        if (top === 0) { winner = (sc.defaultChoice && s.choices.includes(sc.defaultChoice)) ? sc.defaultChoice : s.choices[s.choices.length - 1]; how = "defaut"; }
        else if (tied.length === 1) winner = tied[0];
        else {
          const chef = this.activePlayers().find(p => this.has(p.id, "chef")), cv = chef && s.votes[chef.id];
          if (S.config.voteMode === "chef" && cv && tied.includes(cv)) { winner = cv; how = "chef"; }
          else { winner = tied[Math.floor(this.rand() * tied.length)]; how = "hasard"; }
        }
      }
      S.stats.votes++;
      const c = sc.choices.find(x => x.id === winner);
      const voters = Object.keys(s.votes).filter(pid => s.votes[pid] === winner);
      this.resolveChoice(c, now, { tally, how, voters });
    }
    resolveChoice(c, now, info) {
      const S = this.st, sc = this.sc();
      if (c.once) S.used[S.scene + ":" + c.id] = true;
      const deltas = {};
      if (c.cost) this.applyFx(c.cost, deltas, "cost");
      if (!c.check && !c.outcome) {   // simple déplacement : pas d'écran de résultat
        S.decisions.push({ scene: sc.title, choice: c.label, success: null, how: info.how });
        S.lastMove = { label: c.label, deltas, tally: info.tally, how: info.how };
        return this.goNext(c.next || sc.next || "END", now);
      }
      let out, success = null, chance = null, helped = false;
      if (c.check) {
        chance = this.chance(c);
        if (c.check.role && info.voters.some(pid => this.has(pid, c.check.role))) { chance += 0.2; helped = true; }
        chance = clamp(chance, 0.05, 0.95);
        success = this.rand() < chance;
        out = this.pick(success ? c.success : c.fail);
        S.stats[success ? "checksOk" : "checksKo"]++;
      } else out = this.pick(c.outcome);
      S.decisions.push({ scene: sc.title, choice: c.label, success, how: info.how });
      this.showOutcome(out || {}, { now, label: c.label, success, chance, helped, helpRole: c.check && c.check.role, tally: info.tally, how: info.how, deltas, next: c.next });
    }
    showOutcome(out, o) {
      const S = this.st, sc = this.sc(), deltas = o.deltas || {};
      this.applyOutcome(out, deltas);
      S.ready = {};
      S.step = {
        kind: "outcome", label: o.label, text: out.text || "", success: o.success == null ? null : o.success, chance: o.chance, helped: !!o.helped, helpRole: o.helpRole || null,
        tally: o.tally || null, how: o.how || null, deltas, next: out.next || o.next || sc.next || "END", until: o.now + this.dur().out, len: this.dur().out, minAt: o.now + 3500, shielded: !!out._shielded
      };
    }
    applyOutcome(out, deltas) {
      const S = this.st; deltas = deltas || {};
      if (out.effects) this.applyFx(out.effects, deltas, "outcome", out);
      (out.flags || []).forEach(f => S.flags[f] = true);
      (out.unflags || []).forEach(f => delete S.flags[f]);
      (out.items || []).forEach(i => { S.items[i] = true; deltas["+" + i] = 1; });
      (out.loseItems || []).forEach(i => delete S.items[i]);
      if (out.log) S.journal.push({ kind: out.log.kind || "note", text: out.log.text || out.log });
      return deltas;
    }
    // Applique des effets sur les ressources, avec bouclier, difficulté et bornes.
    applyFx(fx, deltas, src, out) {
      const S = this.st, D = this.diff(), m = this.M();
      for (const k of Object.keys(fx)) {
        if (!(k in RES)) continue;
        let v = fx[k];
        if (k === "integrite" && v < 0) {
          if (S.shield && src !== "cost") { S.shield = false; if (out) out._shielded = true; deltas.bouclier = 1; continue; }
          v = Math.round(v * D.dmg);
        }
        const before = S.res[k];
        let after = clamp(before + v, k === "integrite" ? -999 : 0, RES_MAX[k]);
        if (src === "event") { if (k === "integrite") after = Math.max(after, Math.min(before, 20)); if (k === "temps") after = Math.max(after, Math.min(before, 1)); }
        if (k === "integrite") after = Math.max(after, 0);
        S.res[k] = after; deltas[k] = (deltas[k] || 0) + (after - before);
        if (k === "energie" && m && m.energyThreat && after - before <= -10) { const b2 = S.res.menace; S.res.menace = clamp(b2 + 1, 0, 5); deltas.menace = (deltas.menace || 0) + (S.res.menace - b2); }
      }
    }
    afterOutcome(now) {
      const S = this.st, s = S.step, next = s.next;
      if (S.res.integrite <= 0) return this.finish(now);
      // événement aléatoire contrôlé (jamais deux scènes de suite, jamais sur une fin)
      const m = this.M(), idx = S.visited.length;
      if (next !== "END" && m.events && m.events.length && idx - S.lastEventScene >= 2 && !S.flags._noEvents) {
        const p = Math.min(0.6, (0.18 + 0.08 * S.res.menace) * this.diff().ev);
        const roll = this.rand();
        if (roll < p) {
          const pool = m.events.filter(e => !S.eventsSeen[e.id] && this.cond(e.cond));
          if (pool.length) {
            const e = pool[Math.floor(this.rand() * pool.length)]; S.eventsSeen[e.id] = true; S.lastEventScene = idx; S.stats.events++;
            const deltas = {}; if (e.effects) this.applyFx(e.effects, deltas, "event");
            (e.flags || []).forEach(f => S.flags[f] = true);
            S.journal.push({ kind: "event", text: e.title });
            S.ready = {}; S.step = { kind: "event", id: e.id, title: e.title, text: e.text, tone: e.tone || "alert", deltas, next, until: now + this.dur().out, len: this.dur().out, minAt: now + 3500 };
            return;
          }
        }
      }
      this.goNext(next, now);
    }
    goNext(next, now) {
      const S = this.st, m = this.M();
      if (next === "$resume") { next = S.resume || "END"; S.resume = null; }
      if (S.res.integrite <= 0) return this.finish(now);
      if (next === "END") return this.finish(now);
      if (S.res.temps <= 0 && m.timeoutScene && !S.visited.includes(m.timeoutScene) && next !== m.timeoutScene) { S.resume = next; return this.enterScene(m.timeoutScene, now); }
      if (m.threatScene && S.res.menace >= (m.threatAt || 4) && !S.visited.includes(m.threatScene) && next !== m.threatScene && !(this.sc(next) || {}).noThreat) { S.resume = next; return this.enterScene(m.threatScene, now); }
      this.enterScene(next, now);
    }
    /* ---------- énigmes ---------- */
    solverOk(pid, pz) {
      if (this.has(pid, pz.solver)) return true;
      return !this.present(pz.solver, true);    // personne d'en ligne avec ce rôle : tout le monde peut répondre
    }
    answer(pid, ans, now) {
      const S = this.st, s = S.step, sc = this.sc();
      if (!s || s.kind !== "puzzle") return { ok: false, reason: "ferme" };
      const pz = sc.puzzle;
      if (!this.solverOk(pid, pz)) return { ok: false, reason: "solveur" };
      let given, good, fb = "";
      if (pz.type === "sequence") {
        if (!Array.isArray(ans)) return { ok: false, reason: "format" };
        given = ans.slice(0, 12).map(String);
        const ids = pz.options.map(o => o.id);
        if (given.length !== pz.answer.length || given.some(x => !ids.includes(x))) return { ok: false, reason: "longueur" };
        const well = given.filter((x, i) => x === pz.answer[i]).length; good = well === pz.answer.length;
        fb = well + " élément" + (well > 1 ? "s" : "") + " sur " + pz.answer.length + " à la bonne place";
      } else if (pz.type === "code") {
        given = normCode(ans).slice(0, 12);
        if (given.length !== String(pz.answer).length) return { ok: false, reason: "longueur" };
        const a = String(pz.answer), well = given.split("").filter((x, i) => x === a[i]).length; good = given === a;
        fb = well + " caractère" + (well > 1 ? "s" : "") + " sur " + a.length + " à la bonne place";
      } else {
        given = String(ans); if (!pz.options.some(o => o.id === given)) return { ok: false, reason: "format" };
        good = given === pz.answer; fb = "Ce n'est pas la bonne réponse";
      }
      if (good) { S.stats.puzzlesOk++; S.decisions.push({ scene: sc.title, choice: "Énigme résolue", success: true }); this.showOutcome(this.pick(pz.success) || {}, { now, label: "Énigme résolue", success: true }); this.bump(); return { ok: true, good: true }; }
      s.attempts--; s.tries.push({ by: S.players[pid].name, given, fb });
      if (pz.failAttempt) { const d = {}; this.applyFx(pz.failAttempt.effects || {}, d, "outcome"); s.lastDeltas = d; }
      if (s.attempts <= 0 || S.res.integrite <= 0) this.puzzleFail(now, "essais");
      this.bump(); return { ok: true, good: false, fb, left: s.attempts };
    }
    puzzleFail(now, why) {
      const S = this.st, pz = this.sc().puzzle; S.stats.puzzlesKo++;
      S.decisions.push({ scene: this.sc().title, choice: "Énigme échouée", success: false });
      this.showOutcome(this.pick(pz.fail) || {}, { now, label: why === "temps" ? "Temps écoulé" : why === "abandon" ? "Énigme abandonnée" : "Plus d'essais", success: false });
    }
    /* ---------- pouvoirs ---------- */
    usePower(pid, pw, now) {
      const S = this.st, p = S.players[pid], P = POWERS[pw], s = S.step;
      if (!P || !p.charges || p.charges[pw] == null) return { ok: false, reason: "role" };
      if (S.phase !== "scene" || !s || !P.steps.includes(s.kind)) return { ok: false, reason: "pas_maintenant" };
      if (p.charges[pw] <= 0) return { ok: false, reason: "epuise" };
      const key = pid + ":" + pw; if (S.powerUse[key]) return { ok: false, reason: "deja" };
      const sc = this.sc(), rv = S.reveals[pid] = S.reveals[pid] || [];
      switch (pw) {
        case "ordre": if (S.step.ordre) return { ok: false, reason: "deja" }; s.ordre = pid; break;
        case "scan": {
          let t = this.txt(sc.scan) || (sc.puzzle && sc.puzzle.scan) || "Aucune anomalie notable : les relevés confirment ce que vous voyez.";
          const ch = this.choicesFor(sc).filter(c => c.check);
          if (ch.length && !sc.puzzle) t += " — Probabilités de réussite : " + ch.map(c => "« " + c.label + " » " + Math.round(clamp(this.chance(c), 0.05, 0.95) * 100) + " %" + (c.check.role ? " (+20 % si un " + BASES[c.check.role].toLowerCase() + " vote pour)" : "")).join(" ; ") + ".";
          rv.push({ icon: P.icon, title: "Scan approfondi", text: t }); break;
        }
        case "memoire": case "traduction": {
          let t;
          if (pw === "memoire") t = this.txt(sc.memory) || (sc.puzzle && sc.puzzle.hint) || "Rien dans vos souvenirs ne correspond à ce lieu. Fiez-vous aux autres spécialistes.";
          else {
            const mine = ROLES[p.role].bases, others = Object.keys(BASES).filter(b => !mine.includes(b));
            const src = sc.puzzle && sc.puzzle.clues ? sc.puzzle.clues : sc.secrets || {};
            const cand = others.filter(b => this.txt(src[b]));
            if (cand.length) { const b = cand[Math.floor(this.rand() * cand.length)]; t = "Information du " + BASES[b].toLowerCase() + " : " + this.txt(src[b]); }
            else t = this.txt(sc.memory) || "Aucune inscription à traduire ici.";
          }
          rv.push({ icon: P.icon, title: P.name === "TRADUCTION" ? "Traduction" : "Mémoire des Anciens", text: t }); break;
        }
        case "recon": {
          const ch = this.choicesFor(sc), list = ch.map(c => "« " + c.label + " » : " + (c.danger || (c.check && this.chance(c) < 0.5) ? "DANGEREUX" : "sans danger visible"));
          rv.push({ icon: P.icon, title: "Reconnaissance", text: (this.txt(sc.recon) ? this.txt(sc.recon) + " — " : "") + list.join(" ; ") + "." }); break;
        }
        case "intervention": S.shield = true; break;
        case "soins": { const b = S.res.integrite; S.res.integrite = Math.min(100, b + 15); if (S.res.integrite === b) return { ok: false, reason: "inutile" }; break; }
        case "reparation": { const b = S.res.energie; S.res.energie = Math.min(100, b + 15); if (S.res.energie === b) return { ok: false, reason: "inutile" }; break; }
      }
      p.charges[pw]--; S.powerUse[key] = true; S.stats.powers++;
      S.announce = { at: now, id: S.seq + 1, pw, icon: P.icon, text: p.name + " utilise " + P.name };
      S.journal.push({ kind: "power", text: p.name + " — " + P.name + " (" + sc.title + ")" });
      this.bump(); return { ok: true };
    }
    /* ---------- fin de mission ---------- */
    finish(now) {
      const S = this.st, m = this.M();
      let e = null;
      if (S.res.integrite <= 0) e = m.endings.find(x => x.type === "critique" && x.dead) || m.endings.find(x => x.type === "critique");
      if (!e) e = m.endings.find(x => !x.dead && this.cond(x.cond)) || m.endings[m.endings.length - 1];
      const sec = (m.secondaries || []).map(x => ({ text: x.text, ok: this.cond(x.cond) }));
      const codex = (m.codex || []).filter(c => this.cond(c.cond)).map(c => ({ id: c.id, title: c.title, text: c.text }));
      const okSec = sec.filter(x => x.ok).length;
      const base = { exceptionnelle: 95, parfaite: 85, partielle: 65, controle: 40, critique: 10 }[e.type];
      const score = clamp(Math.round(base + (S.res.integrite - 50) / 10 + okSec * 3 + S.stats.puzzlesOk * 2 - S.stats.puzzlesKo * 2), 0, 100);
      const grade = score >= 92 ? "S" : score >= 80 ? "A" : score >= 62 ? "B" : score >= 40 ? "C" : "D";
      S.phase = "ending";
      S.step = { kind: "ending" };
      S.ending = { type: e.type, title: e.title, text: e.text, name: ENDINGS[e.type].name, res: Object.assign({}, S.res), sec, codex, score, grade, decisions: S.decisions.slice(-12), stats: Object.assign({}, S.stats), minutes: Math.max(1, Math.round((now - S.startedAt) / 60000)), mission: m.id, unlocks: [] };
      if (S.prog) S.ending.unlocks = applyProgress(S.prog, S.ending, m);
      this.bump();
    }

    scenePublic() {
      const S = this.st, sc = this.sc(); if (!sc) return null;

    }
    publicStep(now) {
      const S = this.st, s = S.step; if (!s) return null;
      const out = { kind: s.kind, remain: this.remain(now), len: s.len || 0 };
      const sc = this.sc();
      if (s.kind === "vote") {
        out.choices = s.choices.map(id => { const c = sc.choices.find(x => x.id === id); return { id, label: c.label, desc: c.desc || "", tags: this.tags(c) }; });
        out.locked = this.lockedChoices(sc).map(c => ({ label: c.label, why: c.lockedWhy || "Indisponible" }));
        out.voted = Object.keys(s.votes).length; out.ordre = !!s.ordre;
      }
      if (s.kind === "puzzle") {
        const pz = sc.puzzle;
        out.puzzle = { type: pz.type, prompt: pz.prompt, display: pz.display, options: pz.options ? pz.options.map(o => ({ id: o.id, label: o.label, icon: o.icon || "" })) : null, length: pz.type === "code" ? String(pz.answer).length : pz.type === "sequence" ? pz.answer.length : 1, attempts: s.attempts, max: s.max, tries: s.tries.map(t => ({ by: t.by, fb: t.fb, given: t.given })), solver: pz.solver, solverName: BASES[pz.solver], holder: this.present(pz.solver, true) ? this.activePlayers().filter(p => p.online && this.has(p.id, pz.solver)).map(p => p.name).join(" ou ") : "toute l'équipe (spécialiste absent)" };
      }
      if (s.kind === "outcome") Object.assign(out, { label: s.label, text: s.text, success: s.success, chance: s.chance == null ? null : Math.round(s.chance * 100), helped: s.helped, helpRole: s.helpRole ? BASES[s.helpRole] : null, tally: s.tally ? Object.keys(s.tally).map(id => ({ label: sc.choices.find(x => x.id === id).label, n: s.tally[id], win: sc.choices.find(x => x.id === id).label === s.label })) : null, how: s.how, deltas: s.deltas, shielded: s.shielded });
      if (s.kind === "event") Object.assign(out, { title: s.title, text: s.text, tone: s.tone, deltas: s.deltas });
      return out;
    }
    tags(c) {

ROLES[p.role].icon : "", want: p.want, ready: !!S.ready[p.id], voted: !!(s && s.votes && s.votes[p.id]) }));

        mission: m ? { code: m.code, name: m.name } : null, res: S.res, remain: this.remain(now), len: S.step && S.step.len || 0
      };
      if (r) v.me.powers = r.powers.map(pw => ({ id: pw, name: POWERS[pw].name, icon: POWERS[pw].icon, desc: POWERS[pw].desc, left: p.charges ? p.charges[pw] : 0, steps: POWERS[pw].steps, usedHere: !!S.powerUse[pid + ":" + pw] }));
      if (S.phase === "roles" && m) v.roleBrief = m.roleBrief ? this.txt(m.roleBrief[r.bases[0]]) || "" : "";
      if (S.phase === "briefing" && m) {
        v.briefing = { lines: m.briefing.lines, objectives: m.briefing.objectives, risks: m.briefing.risks };
        v.privateBrief = r.bases.map(b => m.briefing.secret && m.briefing.secret[b] ? { role: BASES[b], text: m.briefing.secret[b] } : null).filter(Boolean);
        v.equip = { chosen: S.equip, choices: this.equipChoices().map(k => ({ id: k, name: EQUIP[k].name, icon: EQUIP[k].icon, desc: EQUIP[k].desc })), mine: r.bases.includes("chef") };
      }
      if (S.phase === "scene") {
        const sc = this.sc(), s = S.step, ps = this.publicStep(now);
        v.scene = { id: S.scene, n: S.visited.length, title: sc.title, step: s.kind, remain: ps.remain, secrets: this.secretsFor(pid), reveals: S.reveals[pid] || [], shield: S.shield };
        if (s.kind === "vote") { v.scene.vote = { choices: ps.choices, mine: s.votes[pid] || null, voted: ps.voted, total: this.online().length, ordre: s.ordre === pid, ordreOn: !!s.ordre }; }
        if (s.kind === "puzzle") { const pz = ps.puzzle; v.scene.puzzle = Object.assign({}, pz, { canAnswer: this.solverOk(pid, sc.puzzle) }); }
        if (s.kind === "outcome" || s.kind === "event") v.scene.result = ps;
      }
      if (S.phase === "ending") v.ending = S.ending;
      return v;
    }
    /* ---------- sauvegarde ---------- */
    save(now) { this.st.savedAt = now; return JSON.stringify(this.st); }
    restore(json, now) {
      let o; try { o = typeof json === "string" ? JSON.parse(json) : json; } catch (e) { return false; }
      if (!o || o.v !== 1 || !o.config || !o.players) return false;
      if (o.mission && !this.missions[o.mission]) return false;
      const prog = this.st.prog; this.st = o; if (prog) this.st.prog = prog;
      Object.values(o.players).forEach(p => p.online = false);
      if (o.phase !== "lobby" && o.phase !== "ending") { o.paused = false; this.pause(now, "restauree"); }
      this.bump(); return true;
    }
  }

  /* ---------- progression (conservée sur la télé) ---------- */
  function newProgress() { return { runs: 0, wins: 0, done: {}, codex: [], equip: [], expert: false }; }
  const RANK = { critique: 0, controle: 1, partielle: 2, parfaite: 3, exceptionnelle: 4 };
  function applyProgress(prog, end, m) {
    const unl = [];
    prog.runs = (prog.runs || 0) + 1; prog.done = prog.done || {}; prog.codex = prog.codex || []; prog.equip = prog.equip || [];
    if (m.training) { if (!prog.done[m.id]) unl.push("Entraînement terminé : vous êtes prêts pour les vraies missions."); prog.done[m.id] = prog.done[m.id] && RANK[prog.done[m.id]] > RANK[end.type] ? prog.done[m.id] : end.type; return unl; }
    const success = ["partielle", "parfaite", "exceptionnelle"].includes(end.type);
    if (success) prog.wins = (prog.wins || 0) + 1;
    const prev = prog.done[m.id]; if (!prev || RANK[end.type] > RANK[prev]) prog.done[m.id] = end.type;
    end.codex.forEach(c => { if (!prog.codex.includes(c.id)) { prog.codex.push(c.id); unl.push("Archive débloquée : " + c.title); } });
    const give = (k) => { if (!prog.equip.includes(k)) { prog.equip.push(k); unl.push("Nouvel équipement : " + EQUIP[k].name); } };
    if (success) give("generateur");
    if (prog.wins >= 2) give("scanner");
    if (end.type === "exceptionnelle" || prog.wins >= 3) give("lexique");
    if (!prog.expert && (end.type === "exceptionnelle" || prog.wins >= 2)) { prog.expert = true; unl.push("Difficulté EXPERT débloquée"); }
    return unl;
  }

  /* ---------- contrôle d'une mission (outil de test) ---------- */
  function checkMission(m) {
    const errs = [], ids = Object.keys(m.scenes);
    const okNext = n => n === "END" || n === "$resume" || m.scenes[n];
    const walkOut = (o, where) => { if (!o) return; (Array.isArray(o) ? o : [o]).forEach(x => { if (x.next && !okNext(x.next)) errs.push(where + " → " + x.next); }); };
    if (!m.scenes[m.startScene]) errs.push("startScene");
    ids.forEach(id => {
      const s = m.scenes[id];
      if (!s.title || !s.text) errs.push(id + " : titre/texte");
      if (s.next && !okNext(s.next)) errs.push(id + ".next → " + s.next);
      (s.choices || []).forEach(c => { if (!c.label) errs.push(id + "." + c.id + " libellé"); if (c.next && !okNext(c.next)) errs.push(id + "." + c.id + " → " + c.next); walkOut(c.outcome, id + "." + c.id); walkOut(c.success, id + "." + c.id + "+"); walkOut(c.fail, id + "." + c.id + "-"); if (c.check && !(c.success && c.fail)) errs.push(id + "." + c.id + " check sans success/fail"); if (!c.check && !c.outcome && !c.next) errs.push(id + "." + c.id + " sans suite"); });
      if (s.puzzle) { const p = s.puzzle; walkOut(p.success, id + " énigme+"); walkOut(p.fail, id + " énigme-"); if (!BASES[p.solver]) errs.push(id + " solveur"); if (p.type === "sequence" && !p.answer.every(a => p.options.some(o => o.id === a))) errs.push(id + " réponse hors options"); if (p.type === "choice" && !p.options.some(o => o.id === p.answer)) errs.push(id + " réponse hors options"); }
      if (!s.puzzle && !(s.choices && s.choices.length) && !s.auto && !s.next) errs.push(id + " : impasse");
      if (s.auto) walkOut(s.auto, id + ".auto");
    });
    ["parfaite", "partielle", "controle", "critique", "exceptionnelle"].forEach(t => { if (!m.endings.some(e => e.type === t)) errs.push("fin manquante " + t); });
    return errs;
  }

  return { Game, ROLES, POWERS, BASES, SLOTS, EQUIP, RES, RES_MAX, ENDINGS, DIFF, DUR, AVATARS, DEFAULT_CONFIG, MIN_PLAYERS, newProgress, applyProgress, checkMission, RANK };
});
