        m.dmg = Math.round(m.dmg * c.dmg * 10) / 10;
        m.r = m.r * (0.85 + 0.15 * c.scale);
      }
      if (m.kind === "grab") m.range = (m.range || 118) * c.grabRange * (0.9 + 0.1 * c.scale);
      m.id = k;
      out[k] = m;
    }
    return out;
  }

  /* ======================= Hasard déterministe ======================= */
  function rng(seed) { let s = (seed >>> 0) || 0x9e3779b9; return () => { s ^= s << 13; s >>>= 0; s ^= s >> 17; s ^= s << 5; s >>>= 0; return s / 4294967296; }; }

  /* ======================= Géométrie des collisions ======================= */
  function segDist2(px, py, ax, ay, bx, by) {
    const dx = bx - ax, dy = by - ay, l = dx * dx + dy * dy;
    let t = l ? ((px - ax) * dx + (py - ay) * dy) / l : 0; t = Math.max(0, Math.min(1, t));
    const x = ax + t * dx - px, y = ay + t * dy - py; return x * x + y * y;
  }
  function hurtShapes(f) {
    const J = f.J, d = f.d, s = [];
    s.push({ k: "head", a: J.head, b: J.head, r: d.head + 4 });
    s.push({ k: "torso", a: J.pelvis, b: J.neck, r: 26 * d.thick });
    s.push({ k: "leg", a: J.pelvis, b: J.knee1, r: 15 * d.thick }); s.push({ k: "leg", a: J.knee1, b: J.foot1, r: 12 * d.thick });
    s.push({ k: "leg", a: J.pelvis, b: J.knee2, r: 15 * d.thick }); s.push({ k: "leg", a: J.knee2, b: J.foot2, r: 12 * d.thick });
    s.push({ k: "arm", a: J.shoulder, b: J.elbow1, r: 11 * d.thick }); s.push({ k: "arm", a: J.shoulder, b: J.elbow2, r: 11 * d.thick });
    return s;
  }
  /* zone de frappe = le segment terminal du membre qui frappe (avant-bras entier + poing, tibia + pied, cuisse + genou) */
  const LIMB_BASE = { fist1: "elbow1", fist2: "elbow2", foot1: "knee1", foot2: "knee2", knee1: "pelvis", shoulder: "pelvis" };
  function hitTest(att, def) {
    const m = att.move, c = att.J[m.limb];
    if (!c) return null;
    const base = att.J[LIMB_BASE[m.limb]] || c, from = m.limb === "shoulder" ? 0.75 : m.limb.startsWith("fist") ? 0 : 0.25;
    const shapes = hurtShapes(def);
    let best = null, bestP = null;
    for (let k = 0; k <= 4; k++) {
      const t = 1 - (1 - from) * (k / 4), px = base[0] + (c[0] - base[0]) * t, py = base[1] + (c[1] - base[1]) * t;
      for (const s of shapes) {
        const rr = m.r + s.r;
        if (segDist2(px, py, s.a[0], s.a[1], s.b[0], s.b[1]) <= rr * rr) { if (!best || s.k === "head") { best = s; bestP = [px, py]; } }
      }
      if (best) break;                     // le point le plus proche de l'extrémité qui touche
    }
    return best ? { x: bestP[0], y: bestP[1], zone: best.k } : null;
  }

  /* ======================= Match ======================= */
  const DEFAULTS = { roundTime: 90, rounds: 3, tie: "prolongation", overtime: 15, dmgScale: 1, training: false };

  function Match(p1, p2, opts) {
    opts = Object.assign({}, DEFAULTS, opts || {});
    const R = rng(opts.seed || 12345);
    const F = [Fighter(p1, 0), Fighter(p2, 1)];
    const S = {
      opts, F, tick: 0, phase: "intro", phaseFr: 0, round: 1, wins: [0, 0], timer: opts.roundTime * FPS,
      hitstop: 0, shake: 0, slow: 0, events: [], winner: null, roundWinner: null, overtime: false, matchFr: 0,
      roundLog: [], camera: { zoom: 1, x: W / 2 }
    };
    const E = (t, o) => S.events.push(Object.assign({ t, tick: S.tick }, o || {}));
    const other = f => F[1 - f.side];
    const grounded = f => f.h <= 0.01;
    function resetRound() {
      F.forEach((f, i) => {
        Object.assign(f, { x: i === 0 ? 700 : 1220, h: 0, vx: 0, vy: 0, facing: i === 0 ? 1 : -1, hp: 100, stam: 100, st: "idle", fr: 0, move: null,
          stun: 0, combo: 0, comboTaken: 0, buf: {}, counterWin: 0, punishWin: 0, inv: 0, seq: null, depth: 0, sp: Math.min(f.sp, 50) });
      });
      S.timer = opts.roundTime * FPS; S.overtime = false; S.phase = "intro"; S.phaseFr = 0; S.roundWinner = null;
    }

    /* ---- entrées ---- */
    function readInput(f, inp) {
      inp = inp || { x: 0, y: 0, b: 0, p: 0 };
      const pressed = (inp.p || 0) | (inp.b & ~f.held);
      f.held = inp.b || 0; f.jx = clamp(inp.x || 0, -1, 1); f.jy = clamp(inp.y || 0, -1, 1);
      for (const k in B) { if (pressed & B[k]) f.buf[k] = BUFFER; else if (f.buf[k] > 0) f.buf[k]--; }
    }
    const clamp = (v, a, b) => Math.max(a, Math.min(b, v));
    const has = (f, k) => f.buf[k] > 0;
    const use = (f, k) => { f.buf[k] = 0; };
    const holding = (f, k) => (f.held & B[k]) !== 0;
    const fwd = f => f.jx * f.facing > 0.35, back = f => f.jx * f.facing < -0.35;
    const down = f => f.jy < -0.55, up = f => f.jy > 0.55;
    const tired = f => f.stam < 15;

    function start(f, st) { f.st = st; f.fr = 0; }
    function startMove(f, id, extra) {
      const m = Object.assign({}, f.moves[id], extra || {});
      if (f.stam < m.stam * 0.5 && m.kind !== "parry") { E("tired", { side: f.side }); }
      f.stam = Math.max(0, f.stam - m.stam);
      f.move = m; f.hitDone = false; f.contact = false; f.armorUsed = false;
      if (tired(f) && m.kind === "strike") m.su += 2;
      start(f, m.kind === "grab" ? "grab" : m.kind === "parry" ? "parry" : "attack");
      E("swing", { side: f.side, move: id, heavy: !!m.heavy });
    }

    const actionable = f => ["idle", "walk", "crouch", "block"].includes(f.st);

    function think(f) {
      const o = other(f);
      // finition / spéciale
      if (has(f, "S") && (actionable(f))) {
        use(f, "S");
        const dist = Math.abs(o.x - f.x);
        if (f.sp >= 100 && dist <= 160 + f.d.half && grounded(o) && ["idle", "walk", "crouch", "block", "attack", "blockstun", "hitstun", "grab", "parry", "stagger"].includes(o.st) && o.st !== "dodge") {
          return startFinisher(f, o);
        }
        if (f.sp >= 50) { f.sp -= 50; startMove(f, "special"); E("special", { side: f.side, name: f.moves.special.label }); return; }
        E("nogauge", { side: f.side });
      }
      if (has(f, "T") && actionable(f)) { use(f, "T"); f.grabBack = back(f); return startMove(f, "grab"); }
      if (has(f, "D") && actionable(f)) {
        use(f, "D");
        if (f.stam >= 10) { f.stam -= 16; f.dodgeKind = back(f) ? "back" : "side"; f.dodgeHit = false; start(f, "dodge"); E("dodge", { side: f.side, kind: f.dodgeKind }); f.stats.dodges++; return; }
        E("tired", { side: f.side });
      }
      // contre après une garde réussie
      if (f.counterWin > 0 && has(f, "P") && actionable(f)) { use(f, "P"); f.counterWin = 0; f.stats.counters++; startMove(f, "counter"); f.move.isCounter = true; E("counter", { side: f.side }); return; }
      // punition après une esquive réussie
      if (f.punishWin > 0 && has(f, "H") && (actionable(f) || (f.st === "dodge" && f.fr > 12))) {
        use(f, "H"); f.punishWin = 0; startMove(f, "cross", { dmg: f.moves.cross.dmg * 1.5, kd: true, su: Math.max(6, f.moves.cross.su - 4), punish: true }); E("punishGo", { side: f.side }); return;
      }
      if (actionable(f)) {
        const cr = down(f);
        if (has(f, "P")) { use(f, "P"); return startMove(f, cr ? "lowjab" : "jab"); }
        if (has(f, "K")) { use(f, "K"); return startMove(f, cr ? "sweep" : fwd(f) ? "stepkick" : "kick"); }
        if (has(f, "H")) { use(f, "H"); return startMove(f, cr ? "upper" : "cross"); }
        if (holding(f, "G")) { if (f.st !== "block") start(f, "block"); f.crouchBlock = cr; return; }
        if (up(f) && !f.prevUp) {
          f.vy = 21 * f.c.jump; f.h = 0.1; f.vx = (fwd(f) ? 6.2 : back(f) ? -5.2 : 0) * f.facing * (0.7 + 0.3 * f.c.speed);
          start(f, "jump"); E("jump", { side: f.side }); return;
        }
        if (cr) { if (f.st !== "crouch") start(f, "crouch"); f.vx = 0; return; }
        if (fwd(f) || back(f)) {
          if (f.st !== "walk") start(f, "walk");
          const sp = (fwd(f) ? 5.4 : -4.4) * f.c.speed * (tired(f) ? 0.75 : 1);
          f.vx = sp * f.facing; return;
        }
        if (f.st !== "idle") start(f, "idle"); f.vx = 0;
      }
      if (f.st === "jump" && !f.move && grounded(f) === false) {
        if (has(f, "K")) { use(f, "K"); f.move = Object.assign({}, f.moves.airk); f.fr = 0; f.hitDone = false; f.stam = Math.max(0, f.stam - f.move.stam); f.airAtk = true; E("swing", { side: f.side, move: "airk" }); }
        else if (has(f, "P")) { use(f, "P"); f.move = Object.assign({}, f.moves.airp); f.fr = 0; f.hitDone = false; f.stam = Math.max(0, f.stam - f.move.stam); f.airAtk = true; E("swing", { side: f.side, move: "airp" }); }
      }
    }

    /* ---- coups portés ---- */
    function applyHit(att, def, hit) {
      const m = att.move;
      att.hitDone = true; att.contact = true;
      // parade de Phantom
      if (def.st === "parry" && def.fr >= def.move.su && def.fr < def.move.su + def.move.ac && m.kind === "strike") {
        E("parry", { side: def.side, x: hit.x, y: hit.y });
        S.hitstop = 10;
        const pm = def.move;
        start(def, "attack"); def.move = Object.assign({}, def.moves.jab, { s: "palmS", c: "parry", su: 2, ac: 4, rc: 14, dmg: pm.dmg, kd: true, push: 140, h: "mid", label: "Reflet" }); def.hitDone = false;
        start(att, "stagger"); att.stun = 26; att.move = null; return;
      }
      // esquive
      if (def.st === "dodge" && def.fr >= 2 && def.fr <= 15) {
        att.hitDone = true; att.contact = false;
        if (!def.dodgeHit) { def.dodgeHit = true; def.punishWin = 32; def.sp = Math.min(100, def.sp + 6); E("evade", { side: def.side, x: hit.x, y: hit.y }); }
        return;
      }
      if (def.inv > 0 || def.st === "down" || def.st === "getup" || def.st === "ko") return;
      const facingAtt = def.facing === -att.facing || Math.abs(def.x - att.x) < 10;
      const blocking = (def.st === "block" || (holding(def, "G") && actionable(def))) && facingAtt;
      const okHeight = !blocking ? false : m.h === "low" ? def.crouchBlock || down(def) : m.h === "over" ? !(def.crouchBlock || down(def)) : true;
      const dmgBase = m.dmg * opts.dmgScale;
      if (blocking && okHeight) {
        const cost = dmgBase * 1.3 * (def.c.blockCost || 1);
        def.stam -= cost; def.stats.blocks++;
        att.combo = 0;
        if (def.stam <= 0) {
          def.stam = 0; start(def, "stagger"); def.stun = 44; def.vx = -att.facing * -4 * 0;
          def.vx = att.facing * 6; E("guardbreak", { side: def.side, x: hit.x, y: hit.y }); S.hitstop = 10; S.shake = 10; return;
        }
        start(def, "blockstun"); def.stun = m.bs; def.vx = att.facing * (m.push / def.c.weight) * 0.12;
        def.sp = Math.min(100, def.sp + 3); att.sp = Math.min(100, att.sp + 1.5);
        E("block", { side: def.side, x: hit.x, y: hit.y, heavy: !!m.heavy });
        S.hitstop = m.heavy ? 6 : 4;
        return;
      }
      // touché
      const comboing = ["hitstun", "stagger"].includes(def.st) || (def.st === "down" && false) || (!grounded(def) && def.st === "juggle");
      if (comboing) att.combo++; else att.combo = 1;
      def.comboTaken = att.combo;
      const scale = Math.max(0.45, 1 - 0.11 * (att.combo - 1));
      let dmg = dmgBase * scale * def.c.taken * (m.heavy || m.kd ? def.c.heavyTaken / def.c.taken : 1);
      if (m.isCounter) dmg *= 1.25;
      if (m.punish) { dmg *= 1; def.stun = 0; }
      if (def.st === "attack" && def.fr < def.move.su) { dmg *= 1.2; E("counterhit", { side: att.side }); }   // interrompt une préparation
      dmg = Math.round(dmg * 10) / 10;
      // armure du Bélier
      if (def.st === "attack" && def.move && def.move.armor && !def.armorUsed && m.light) { def.armorUsed = true; def.hp -= dmg * 0.5; E("armor", { side: def.side, x: hit.x, y: hit.y }); S.hitstop = 5; return; }
      def.hp = Math.max(0, def.hp - dmg);
      att.stats.hits++; att.stats.dmg += dmg; att.stats.maxCombo = Math.max(att.stats.maxCombo, att.combo);
      if (att.combo === 2) att.stats.combos++;
      if (m.punish) att.stats.punishes++;
      att.sp = Math.min(100, att.sp + dmg * 1.3); def.sp = Math.min(100, def.sp + dmg * 0.8);
      def.move = null; def.airAtk = false;
      const hs = Math.max(8, Math.round(m.hs * (1 - 0.12 * (att.combo - 1))));
      const push = (m.push / def.c.weight) * (1 + 0.15 * (att.combo - 1));
      const big = m.heavy || m.kd || dmg >= 12;
      E("hit", { side: def.side, by: att.side, x: hit.x, y: hit.y, dmg, zone: hit.zone, heavy: big, combo: att.combo, move: m.id, label: m.label, punish: !!m.punish, counter: !!m.isCounter });
      if (att.combo >= 2) E("combo", { side: att.side, n: att.combo });
      S.hitstop = big ? 9 : 5; S.shake = big ? 9 : 3;
      if (def.hp <= 0) return knockout(att, def, hit);
      if (m.kd || (!grounded(def))) {
        start(def, "down"); def.downKind = m.id === "sweep" ? "sweep" : "back"; def.vx = att.facing * Math.max(4, push * 0.09); def.vy = m.antiair || !grounded(def) ? 9 : 4; def.h = Math.max(def.h, 1);
        att.combo = att.combo; return;
      }
      start(def, "hitstun"); def.stun = hs; def.hitZone = hit.zone === "head" ? "high" : m.h === "low" ? "low" : "body";
      def.vx = att.facing * push * 0.11;
    }

    function knockout(att, def, hit) {
      start(def, "ko"); def.vx = att.facing * 7; def.vy = 10; def.h = Math.max(def.h, 1);
      S.hitstop = 14; S.slow = 70; S.shake = 18;
      E("ko", { side: def.side, by: att.side, x: hit ? hit.x : def.x, y: hit ? hit.y : GROUND - 200 });
      endRound(att.side, "ko");
    }

    function endRound(winner, how) {
      if (S.phase !== "fight") return;
      S.phase = "roundEnd"; S.phaseFr = 0; S.roundWinner = winner; S.roundHow = how;
      if (winner === 0 || winner === 1) S.wins[winner]++;
      S.roundLog.push({ round: S.round, winner, how, time: Math.round((opts.roundTime * FPS - S.timer) / FPS), hp: F.map(f => Math.round(f.hp)) });
      E("roundEnd", { winner, how });
    }

    /* ---- prises ---- */
    function tryGrab(f) {
      const o = other(f), m = f.move;
      if (f.fr < m.su || f.fr >= m.su + m.ac || f.hitDone) return;
      const dist = Math.abs(o.x - f.x) - f.d.half - o.d.half;
      const okState = grounded(o) && !["down", "getup", "ko", "dodge", "thrown", "fin_v", "jump"].includes(o.st) && o.inv <= 0;
      if (dist <= m.range - 60 && okState && Math.sign(o.x - f.x || f.facing) === f.facing) {
        f.hitDone = true;
        const kind = m.throwKind === "spin" ? "throwSpin" : (f.grabBack || back(f)) ? "throwB" : "throwF";
        beginSeq(f, o, kind, (m.throwDmg || 12) * f.c.throwDmg * opts.dmgScale);
        f.stats.throws++;
        E("grab", { side: f.side, kind });
      }
    }
    function beginSeq(att, vic, kind, dmgTotal) {
      const sq = SEQ[kind];
      att.seq = { kind, sq, vic: vic.side, dmgTotal, fin: kind.startsWith("fin_") }; att.seqFr = 0;
      start(att, att.seq.fin ? "fin" : "throw"); start(vic, att.seq.fin ? "fin_v" : "thrown");
      vic.seqBy = att.side; att.move = null; vic.move = null; att.vx = 0; vic.vx = 0; vic.h = 0; att.h = 0;
      att.combo = 0;
    }
    function startFinisher(f, o) {
      f.sp = 0; f.stats.finishers++;
      beginSeq(f, o, "fin_" + f.cid, 30);
      E("finisher", { side: f.side, name: f.c.finisher.name });
      S.slow = 0;
    }
    function runSeq(att) {
      const v = F[att.seq.vic], sq = att.seq.sq, fr = att.seqFr;
      // dégagement : la victime appuie sur PRISE dans les 8 premières images (projections simples seulement)
      if (!att.seq.fin && att.seq.kind !== "throwSpin" && fr < 9 && has(v, "T")) {
        use(v, "T"); att.seq = null; v.seqBy = null;
        start(att, "blockstun"); att.stun = 14; att.vx = -att.facing * 7;
        start(v, "blockstun"); v.stun = 14; v.vx = att.facing * 7;
        E("break", { side: v.side }); return;
      }
      sq.hits.forEach(h => {
        if (h.f === fr) {
          const share = att.seq.fin ? h.dmg : att.seq.dmgTotal * (h.dmg / sq.hits.reduce((a, b) => a + b.dmg, 0));
          const dmg = Math.round(share * v.c.taken * 10) / 10;
          v.hp = Math.max(0, v.hp - dmg); att.stats.dmg += dmg; att.stats.hits++;
          att.sp = Math.min(100, att.sp + (att.seq.fin ? 0 : dmg)); v.sp = Math.min(100, v.sp + dmg * 0.6);
          const J = v.J || {}; const px = J.pelvis ? J.pelvis[0] : v.x, py = J.pelvis ? J.pelvis[1] : GROUND - 80;
          E("slam", { side: v.side, by: att.side, x: px, y: py, dmg, heavy: h.shake >= 10, fin: att.seq.fin });
          S.shake = h.shake; S.hitstop = h.shake >= 10 ? 8 : 3;
          if (h.slow) S.slow = h.slow;
        }
      });
      att.seqFr++;
      if (att.seqFr >= sq.len) {
        const swap = sq.swap;
        att.seq = null; v.seqBy = null;
        start(att, "idle");
        if (v.hp <= 0) { start(v, "ko"); v.koLying = true; E("ko", { side: v.side, by: att.side, x: v.x, y: GROUND - 60 }); S.slow = Math.max(S.slow, 60); endRound(att.side, "ko"); }
        else { start(v, "down"); v.fr = 18; v.downKind = swap ? "swap" : "back"; }
        if (swap) att.facing = -att.facing;
      }
    }

    /* ---- avance d'un combattant ---- */
    function update(f) {
      const o = other(f);
      f.fr++;
      if (f.counterWin > 0) f.counterWin--;
      if (f.punishWin > 0) f.punishWin--;
      if (f.inv > 0) f.inv--;
      switch (f.st) {
        case "attack": {
          const m = f.move;
          if (!m) { start(f, "idle"); break; }
          if (m.dash && f.fr <= m.su + m.ac) f.vx = m.dash * f.facing * (f.fr > m.su - 4 ? 1 : 0.3); else f.vx *= 0.8;
          if (m.hop && f.fr === m.su - 2) { f.vy = m.hop; f.h = 0.1; }
          const end = m.su + m.ac + m.rc;
          // enchaînements : après contact, pendant la récupération
          if (f.contact && f.fr > m.su && f.fr < end - 2) {
            for (const k of ["P", "K", "H"]) if (m.chain && m.chain[k] && has(f, k)) {
              use(f, k); const nx = m.chain[k]; startMove(f, nx); E("chain", { side: f.side, move: nx }); return;
            }
          }
          if (f.fr >= end) { f.move = null; start(f, "idle"); }
          break;
        }
        case "grab": {
          const m = f.move; tryGrab(f);
          if (f.st === "grab" && f.fr >= m.su + m.ac + m.rc) { f.move = null; start(f, "idle"); }
          if (f.st === "grab" && f.fr === m.su + m.ac && !f.hitDone) E("whiff", { side: f.side, move: "grab" });
          break;
        }
        case "parry": { const m = f.move; if (f.fr >= m.su + m.ac + m.rc) { f.move = null; start(f, "idle"); } break; }
        case "block": if (!holding(f, "G")) start(f, "idle"); f.crouchBlock = down(f); f.vx = 0; break;
        case "blockstun": f.vx *= 0.82; if (f.fr >= f.stun) { start(f, holding(f, "G") ? "block" : "idle"); f.counterWin = f.c.counterWin || 14; } break;
        case "hitstun": f.vx *= 0.85; if (f.fr >= f.stun) { start(f, "idle"); f.comboTaken = 0; o.combo = 0; } break;
        case "stagger": f.vx *= 0.85; if (f.fr >= f.stun) { start(f, "idle"); } break;
        case "dodge": {
          const len = 20;
          if (f.dodgeKind === "back") f.vx = f.fr < 12 ? -7.5 * f.facing : 0; else f.vx = 0;
          f.depth = f.dodgeKind === "side" ? Math.sin(Math.min(1, f.fr / len) * Math.PI) : 0;
          if (f.fr >= len) { f.depth = 0; start(f, "idle"); }
          break;
        }
        case "down": {
          if (!grounded(f)) break;
          f.vx *= 0.86;
          if (f.fr === 20) E("land", { side: f.side, x: f.x });
          if (f.fr >= 64) { start(f, "getup"); f.inv = 26; }
          break;
        }
        case "getup": f.vx = 0; if (f.fr >= 24) { start(f, "idle"); f.inv = 6; } break;
        case "ko": if (grounded(f)) f.vx *= 0.86; if (f.fr === 22) E("land", { side: f.side, x: f.x, ko: true }); break;
        case "jump":
          if (f.move) { if (f.fr > f.move.su + f.move.ac + f.move.rc) { f.move = null; } }
          break;
        case "throw": case "fin": runSeq(f); break;
      }
    }

    function physics(f) {
      if (f.st === "thrown" || f.st === "fin_v" || f.st === "throw" || f.st === "fin") return;
      if (f.h > 0 || f.vy > 0) {
        f.h += f.vy; f.vy -= GRAVITY;
        if (f.h <= 0) {
          f.h = 0; f.vy = 0;
          if (f.st === "jump") { start(f, "idle"); f.move = null; f.airAtk = false; E("land", { side: f.side, x: f.x, soft: true }); f.vx = 0; }
          else if (f.st === "attack" && f.move && f.move.hop) { /* le genou retombe */ }
        }
      }
      f.x += f.vx;
    }

    function separate() {
      const [a, b] = F;
      const busy = s => ["throw", "thrown", "fin", "fin_v"].includes(s);
      if (busy(a.st) || busy(b.st)) return;
      const overlapY = Math.abs(a.h - b.h) < 150;
      const minD = a.d.half + b.d.half;
      const dx = b.x - a.x;
      if (overlapY && Math.abs(dx) < minD && a.depth < 0.5 && b.depth < 0.5) {
        const push = (minD - Math.abs(dx)) / 2, s = dx === 0 ? (a.facing > 0 ? 1 : -1) : Math.sign(dx);
        a.x -= push * s; b.x += push * s;
      }
      for (const f of F) f.x = clamp(f.x, LEFT + f.d.half, RIGHT - f.d.half);
      // si l'un est bloqué contre la corde, l'autre recule
      if (overlapY && Math.abs(b.x - a.x) < minD - 1 && a.depth < 0.5 && b.depth < 0.5) {
        const s = Math.sign(b.x - a.x) || 1;
        if (a.x <= LEFT + a.d.half + 0.5 || a.x >= RIGHT - a.d.half - 0.5) b.x = a.x + s * minD; else a.x = b.x - s * minD;
        for (const f of F) f.x = clamp(f.x, LEFT + f.d.half, RIGHT - f.d.half);
      }
    }

    function autoFace(f) {
      const o = other(f);
      if (!grounded(f)) return;
      if (!["idle", "walk", "crouch", "block"].includes(f.st)) return;
      const want = o.x >= f.x ? 1 : -1;
      if (want !== f.facing && Math.abs(o.x - f.x) > 12) { f.facing = want; f.turned = 6; }
    }

    /* ---- pose de chaque état ---- */
    function poseOf(f) {
      const m = f.move, fr = f.fr;
      const breath = Math.sin((S.tick + f.side * 40) / 14);
      switch (f.st) {
        case "idle": { const g = lerpPose(P("guard"), P("guard2"), (breath + 1) / 2); return g; }
        case "walk": { const t = (S.tick * 0.11 * f.c.speed) % 4; const ks = [P("walkA"), P("walkB"), P("walkC"), P("walkB"), P("walkA")]; const i = Math.floor(t); return lerpPose(ks[i], ks[i + 1], ease(t - i)); }
        case "crouch": return P("crouch");
        case "block": return f.crouchBlock ? P("cblock") : P("block");
        case "blockstun": return lerpPose(f.crouchBlock ? P("cblock") : P("block"), Object.assign({}, f.crouchBlock ? P("cblock") : P("block"), { lean: -6, hd: -18 }), Math.sin(Math.min(1, fr / Math.max(1, f.stun)) * Math.PI));
        case "hitstun": { const hp = f.hitZone === "high" ? P("hitHigh") : f.hitZone === "low" ? P("hitLow") : P("hitBody"); return anim([[0, P("guard")], [3, hp], [Math.max(4, f.stun - 4), hp], [f.stun, P("guard")]], fr); }
        case "stagger": return anim([[0, P("hitHigh")], [6, P("stagger")], [Math.max(7, f.stun - 6), P("stagger")], [f.stun, P("guard")]], fr);
        case "jump": {
          if (m) { const end = m.su + m.ac; return anim([[0, P("jump")], [m.su, P(m.s)], [end, P(m.s)], [end + m.rc, P("jump")]], fr); }
          return f.vy > 4 ? P("jump") : lerpPose(P("jump"), P("fall"), clamp((4 - f.vy) / 18, 0, 1));
        }
        case "attack": case "grab": case "parry": {
          if (!m) return P("guard");
          const base = m.crouch ? P("crouch") : P("guard");
          const c = P(m.c), s = P(m.s), su = m.su, act = su + m.ac, end = act + m.rc;
          return anim([[0, base], [Math.max(1, Math.round(su * 0.6)), c], [su, s], [act, s], [end, base]], fr);
        }
        case "dodge": { const p = f.dodgeKind === "back" ? P("dodgeBack") : P("dodgeSide"); return anim([[0, P("guard")], [5, p], [14, p], [20, P("guard")]], fr); }
        case "down": {
          if (f.downKind === "sweep") return anim([[0, P("hitLow")], [10, Object.assign({}, P("down"), { rot: 60 })], [20, P("down")]], fr);
          return anim([[0, P("hitHigh")], [12, Object.assign({}, P("down"), { rot: 70 })], [20, P("down")], [60, P("down")], [64, P("getup")]], fr);
        }
        case "getup": return anim([[0, P("getup")], [16, P("crouch")], [24, P("guard")]], fr);
        case "ko": return anim([[0, P("hitHigh")], [14, Object.assign({}, P("down"), { rot: 70 })], [22, P("down")]], fr);
        case "win": return anim([[0, P("guard")], [20, P("victory")], [80, P("victory")], [110, P("victory2")], [160, P("victory")]], fr % 160);
        case "lose": return P("down");
        case "throw": case "fin": { const sq = f.seq.sq; const fr2 = f.seqFr; return seqPose(sq, fr2, "a"); }
        case "thrown": case "fin_v": { const a = F[f.seqBy]; if (!a || !a.seq) return P("down"); return seqPose(a.seq.sq, a.seqFr, "v"); }
      }
      return P("guard");
    }
    function seqPose(sq, fr, who) {
      const ks = sq.keys.map(k => [k.f, typeof k[who] === "string" ? P(k[who]) : k[who]]);
      return anim(ks, fr);
    }
    function seqPlace(att) {               // place la victime d'une séquence
      const sq = att.seq.sq, v = F[att.seq.vic], fr = att.seqFr;
      const ks = sq.keys;
      let dx = ks[0].dx, dh = ks[0].dh, ah = 0, spin = 0;
      for (let i = 1; i < ks.length; i++) if (fr <= ks[i].f) {
        const t = ease((fr - ks[i - 1].f) / Math.max(1, ks[i].f - ks[i - 1].f));
        dx = ks[i - 1].dx + (ks[i].dx - ks[i - 1].dx) * t; dh = ks[i - 1].dh + (ks[i].dh - ks[i - 1].dh) * t;
        ah = (ks[i - 1].ah || 0) + ((ks[i].ah || 0) - (ks[i - 1].ah || 0)) * t;
        spin = ks[i].spin ? ks[i].spin * t : 0; break;
      } else if (i === ks.length - 1) { dx = ks[i].dx; dh = ks[i].dh; ah = ks[i].ah || 0; }
      att.h = ah;
      // tournoiement : l'attaquant et la victime changent de côté visuellement
      att.spinView = spin ? Math.cos(spin * Math.PI * 2) : 1;
      const side = att.facing * (spin ? att.spinView : 1);
      v.x = clamp(att.x + side * dx, LEFT - 60, RIGHT + 60); v.h = dh; v.facing = -att.facing * (spin ? Math.sign(att.spinView) || 1 : 1);
      v.spinView = att.spinView;
    }

    function computeJ(f) {
      f.pose = poseOf(f);
      const Jl = fkLocal(f.d, f.pose);
      // en l'air ou porté : hauteur de bassin de référence (debout) pour ne pas « recaler » le corps
      const ref = (f.st === "thrown" || f.st === "fin_v") ? 0 : null;
      if (ref === 0) f.J = toWorld(Jl, f.x, f.h, f.facing);
      else if (!grounded(f) && f.st !== "down" && f.st !== "ko") f.J = toWorld(Jl, f.x, f.h, f.facing, STANDLIFT(f));
      else f.J = toWorld(Jl, f.x, f.h, f.facing);
    }
    const standCache = {};
    function STANDLIFT(f) { if (!standCache[f.cid]) standCache[f.cid] = fkLocal(f.d, P0).lift; return standCache[f.cid]; }

    /* ======================= un pas de simulation ======================= */
    function step(inputs) {
      S.events = [];
      S.tick++;
      inputs = inputs || [];
      F.forEach((f, i) => readInput(f, inputs[i]));
      if (S.slow > 0) S.slow--;
      if (S.shake > 0) S.shake *= 0.86;
      if (S.shake < 0.3) S.shake = 0;
      if (S.hitstop > 0) { S.hitstop--; F.forEach(f => { f.prevUp = up(f); }); return S; }
      S.phaseFr++;
      if (S.phase === "intro") {
        F.forEach(f => { if (f.st !== "idle") start(f, "idle"); f.vx = 0; });
        if (S.phaseFr === 1) E("round", { n: S.round, final: S.wins[0] === S.wins[1] && S.wins[0] === Math.floor(opts.rounds / 2) });
        if (S.phaseFr === 90) E("fight", {});
        if (S.phaseFr >= 110) { S.phase = "fight"; S.phaseFr = 0; }
      }
      const fighting = S.phase === "fight";
      F.forEach(f => {
        if (fighting) think(f);
        f.prevUp = up(f);
      });
      F.forEach(f => update(f));
      F.forEach(f => physics(f));
      separate();
      F.forEach(f => autoFace(f));
      F.forEach(f => { if (f.st === "throw" || f.st === "fin") { if (f.seq) seqPlace(f); } });
      F.forEach(f => computeJ(f));
      // collisions des coups
      for (const att of F) {
        const m = att.move;
        if (!m || att.hitDone || m.kind !== "strike") continue;
        const inAir = att.st === "jump";
        if (!(att.st === "attack" || inAir)) continue;
        if (att.fr < m.su || att.fr >= m.su + m.ac) { if (att.fr === m.su + m.ac && !att.hitDone && !att.contact) { E("whiff", { side: att.side, move: m.id, heavy: !!m.heavy }); } continue; }
        const def = other(att);
        if (def.st === "throw" || def.st === "thrown" || def.st === "fin" || def.st === "fin_v") continue;
        if (def.depth > 0.4 && def.st === "dodge") { const h = hitTest(att, def); if (h) applyHit(att, def, h); continue; }
        const h = hitTest(att, def);
        if (h) applyHit(att, def, h);
      }
      // jauges
      F.forEach(f => {
        const resting = ["idle", "walk", "crouch"].includes(f.st);
        const rate = (resting ? 0.32 : f.st === "block" ? 0.06 : 0.12) * f.c.regen;
        f.stam = Math.min(100, f.stam + rate);
      });
      // chronomètre
      if (fighting) {
        S.timer--;
        if (S.timer <= 0) {
          const [a, b] = F;
          if (Math.abs(a.hp - b.hp) < 0.05) {
            if (opts.tie === "prolongation" && !S.overtime) { S.overtime = true; S.timer = opts.overtime * FPS; E("overtime", {}); }
            else { E("timeup", { draw: true }); endRound(-1, "draw"); }
          } else { E("timeup", {}); endRound(a.hp > b.hp ? 0 : 1, "time"); }
        }
      }
      if (S.phase === "roundEnd") {
        if (S.phaseFr === 60) F.forEach(f => { if (S.roundWinner === f.side && f.st !== "ko") { start(f, "win"); f.vx = 0; } });
        if (S.phaseFr >= 210) {
          const need = Math.floor(opts.rounds / 2) + 1;
          if (S.wins[0] >= need || S.wins[1] >= need || (opts.training && false)) {
            S.phase = "matchEnd"; S.phaseFr = 0; S.winner = S.wins[0] > S.wins[1] ? 0 : 1;
            F.forEach(f => { if (f.side === S.winner) start(f, "win"); });
            E("matchEnd", { winner: S.winner });
          } else { S.round++; resetRound(); }
        }
      }
      if (S.phase === "fight" || S.phase === "roundEnd") S.matchFr++;
      return S;
    }

    resetRound();
    F.forEach(f => computeJ(f));
    return { step, S, F, view: () => S };
  }

  /* ======================= IA d'entraînement =======================
     Décide à partir de la situation : distance, état de l'adversaire, jauges. Délai de réaction selon le niveau. */
  let aiCount = 0;
  function AI(level, seed) {
    const L = { facile: { react: 22, block: 0.35, punish: 0.3, aggr: 0.25 }, normal: { react: 14, block: 0.6, punish: 0.6, aggr: 0.45 }, difficile: { react: 8, block: 0.85, punish: 0.9, aggr: 0.6 } }[level || "normal"];
    const r = rng(seed != null ? seed : 777 + 97 * (aiCount++));
    let plan = null, wait = 0, held = 0, seen = [];
    const out = { x: 0, y: 0, b: 0, p: 0 };
    return function (S, me) {
      const f = S.F[me], o = S.F[1 - me];
      seen.push({ st: o.st, fr: o.fr, move: o.move && o.move.id, x: o.x });
      if (seen.length > 40) seen.shift();
      const past = seen[Math.max(0, seen.length - 1 - L.react)] || seen[0];
      out.p = 0; out.b = 0; out.x = 0; out.y = 0;
      if (S.phase !== "fight") return out;
      const dist = Math.abs(o.x - f.x) - f.d.half - o.d.half;
      const toward = Math.sign(o.x - f.x) || f.facing;
      const want = f.cid === "titan" || f.cid === "atlas" ? 30 : f.cid === "volt" ? 70 : 90;
      if (wait > 0) { wait--; if (plan === "block") out.b |= B.G; if (plan === "away") out.x = -toward; return out; }
      plan = null;
      const threat = past && (past.st === "attack" && past.move !== null) && dist < 190;
      if (threat && r() < L.block) {
        if (r() < 0.25 && f.stam > 30) { out.p |= B.D; wait = 12; return out; }
        plan = "block"; out.b |= B.G; wait = 10; return out;
      }
      // punir une attaque ratée
      if ((o.st === "attack" || o.st === "grab") && o.move && o.fr > o.move.su + o.move.ac && dist < 120 && r() < L.punish) { out.p |= B.H; return out; }
      if (f.punishWin > 0) { out.p |= B.H; return out; }
      if (f.counterWin > 0 && dist < 110) { out.p |= B.P; return out; }
      if (o.st === "down" || o.st === "getup") { if (dist > 140) out.x = toward; return out; }
      if (f.sp >= 100 && dist < 120) { out.p |= B.S; return out; }
      if (f.sp >= 50 && dist > 120 && dist < 320 && r() < 0.03) { out.p |= B.S; return out; }
      if (o.st === "jump" && dist < 160 && r() < 0.5) { out.y = -1; out.p |= B.H; return out; }
      if (dist > want + 40) { out.x = toward; if (dist > 380 && r() < 0.01) out.y = 1; return out; }
      if (f.st === "attack" && f.contact) {   // enchaînement
        if (f.move.id === "jab") out.p |= r() < 0.6 ? B.P : B.K; else if (f.move.id === "hook") out.p |= B.K;
        return out;
      }
      if (dist < 70 && (o.st === "block" || o.st === "blockstun") && r() < 0.25) { out.p |= B.T; return out; }
      if (dist < 140 && r() < L.aggr * 0.25) { const c = r(); out.p |= c < 0.45 ? B.P : c < 0.75 ? B.K : c < 0.9 ? B.H : B.T; return out; }
      if (dist < want - 20 && r() < 0.05) { plan = "away"; wait = 10; out.x = -toward; return out; }
      if (r() < 0.02) out.y = -1;
      return out;
    };
  }
