function wolvesFor(n){return n<=5?1:n<=8?2:n<=10?2:3}
function autoDeck(n,level){
  const L=LEVELS[level]||LEVELS.debutant;
  const deck={};let w=wolvesFor(n);
  const specials=L.pick.slice();
  // nombre de rôles spéciaux : on garde au moins 1 villageois simple à partir de 7 joueurs
  let room=n-w-(n>=7?1:0);
  for(const r of specials){
    if(room<=0)break;
    if(WOLF_ROLES.includes(r)){deck[r]=1;w--;continue}  // un loup spécial remplace un loup simple
    deck[r]=1;room--;
  }
  if(w>0)deck.loup=w;
  const used=Object.values(deck).reduce((a,b)=>a+b,0);
  if(n-used>0)deck.villageois=n-used;
  return deck;
}

/* Vérifie une composition. Renvoie {errors:[], warnings:[], fixes:[]} */
function validate(deck,n){
  const errors=[],warnings=[],fixes=[];
  const need=n+(deck.voleur?2:0);
  const size=deckSize(deck);
  Object.keys(deck).forEach(r=>{if(!ROLES[r])errors.push("Rôle inconnu : "+r);else if(ROLES[r].impl===false&&deck[r])errors.push(ROLES[r].n+" n'est pas encore jouable.")});
  Object.keys(deck).forEach(r=>{const R=ROLES[r];if(R&&!R.multi&&!R.group&&deck[r]>1)errors.push("Un seul "+R.n+" par partie.")});
  if(deck.soeur&&deck.soeur!==2){errors.push("Les Sœurs vont toujours par deux.");fixes.push("Mettez exactement 2 Sœurs.")}
  if(deck.frere&&deck.frere!==3){errors.push("Les Frères vont toujours par trois.");fixes.push("Mettez exactement 3 Frères.")}
  if(n<4)errors.push("Il faut au moins 4 joueurs.");
  if(n>10)errors.push("10 joueurs maximum.");
  if(size!==need){
    errors.push("La composition compte "+size+" carte"+(size>1?"s":"")+" pour "+need+" nécessaire"+(need>1?"s":"")+(deck.voleur?" (joueurs + 2 cartes du Voleur)":"")+".");
    fixes.push(size<need?"Ajoutez "+(need-size)+" Villageois.":"Retirez "+(size-need)+" carte"+(size-need>1?"s":"")+".");
  }
  const wolves=WOLF_ROLES.reduce((a,r)=>a+(deck[r]||0),0);
  const wolvesish=wolves+(deck.enfantsauvage?1:0)+(deck.chienloup?1:0);
  if(wolves===0&&!deck.chienloup&&!deck.enfantsauvage){errors.push("Il faut au moins un loup-garou.");fixes.push("Ajoutez 1 Loup-Garou.")}
  else if(wolves===0)warnings.push("Aucun loup au départ : la partie dépendra du Chien-Loup ou de l'Enfant Sauvage.");
  if(n>=4&&wolves>=n/2){errors.push("Trop de loups : ils gagneraient dès le premier soir.");fixes.push("Retirez "+(wolves-Math.ceil(n/2)+1)+" loup(s).")}
  else if(n>=4&&wolvesish/n>0.4)warnings.push("Composition déséquilibrée en faveur des loups.");
  else if(n>=6&&wolves/n<0.15)warnings.push("Composition déséquilibrée en faveur du village.");
  if(deck.loupblanc&&wolves<2)warnings.push("Le Loup Blanc est plus intéressant avec au moins un autre loup.");
  if(deck.flute&&n<6)warnings.push("Le Joueur de flûte gagne très vite à moins de 6 joueurs.");
  if(deck.ange&&n<6)warnings.push("L'Ange est conseillé à partir de 6 joueurs.");
  if(deck.petitefille&&!wolves)warnings.push("La Petite Fille n'a personne à espionner sans loup.");
  if(deck.gml&&wolves<2)warnings.push("Le Grand Méchant Loup chasse mieux avec au moins un autre loup.");
  return {errors,warnings,fixes};
}

  function tick(){
    if(S.paused)return;
    if(S.phase==="night"&&S.step&&S.stepAt&&io.now()-S.stepAt>STEP_LIMIT&&actors(S.step).length){journal("Temps écoulé : "+S.step);endStep();return}

    if(S.phase==="day"&&io.now()>=S.dayEnd)startVote(false);
    else if(S.phase==="vote"&&io.now()>=S.voteEnd)resolveVote();
  }

  }

/* Décision d'un robot (mode démonstration et tests) : renvoie une action ou null. */
function botDecide(v,rand){
  const pr=v&&v.p;if(!pr)return null;
  const c=pr.choices||[];const pick=()=>c[Math.floor(rand()*c.length)];
  switch(pr.k){
    case "reveal":return pr.ready?null:{t:"ready"};
    case "thief":return {t:"thief",c:pr.must||rand()<0.5?Math.floor(rand()*2):null};
    case "cupid":if(c.length<2)return null;{const a=pick();let b=pick();let g=0;while(b.i===a.i&&g++<20)b=pick();return {t:"cupid",a:a.i,b:b.i}}
    case "lovers":return pr.done?null:{t:"ok"};
    case "result":return {t:"ok"};
    case "choice":return {t:"choose",v:pr.options[Math.floor(rand()*pr.options.length)].v};
    case "pick":if(pr.optional&&rand()<0.25)return {t:"pass"};return c.length?{t:"pick",target:pick().i}:(pr.optional?{t:"pass"}:null);
    case "wolves":{if(pr.mine!=null||!c.length)return null;const lead=pr.pack.find(w=>w.t);const t=lead?c.find(x=>x.n===lead.t):null;return {t:"pick",target:(t||pick()).i}}
    case "spy":return !pr.used&&rand()<0.4?{t:"spy"}:null;
    case "witch":return {t:"witch",save:pr.life&&!!pr.victim&&rand()<0.5,kill:pr.death&&rand()<0.25&&c.length?pick().i:null};
    case "flute":{if(!c.length)return null;const a=pick();let b=pick();let g=0;while(c.length>1&&b.i===a.i&&g++<20)b=pick();return {t:"flute",a:a.i,b:b.i}}
    case "day":return pr.asked?null:{t:"tovote"};
    case "vote":if(pr.mine!=null)return null;if(pr.judge&&rand()<0.2)return {t:"judge"};if(pr.abstain&&rand()<0.1)return {t:"abstain"};return c.length?{t:"pick",target:pick().i}:null;
  }
  return null;
}
